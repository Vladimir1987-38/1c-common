# Стандарты разработки группы разработки

Обязательны для всех подключённых проектов наравне с [общими стандартами 1С](1c-development.md). Перед каждым новым этапом изменения кода читать актуальную редакцию. Перед завершением проверять изменённые фрагменты по всем применимым правилам ниже.

Примеры иллюстрируют отдельные требования к оформлению и могут быть неполными фрагментами. Предметные значения и формулы из примеров не переносить в рабочий код без требований задачи. Пометки «НЕПРАВИЛЬНО» показывают варианты, которых следует избегать.

## 1. Выравнивание общих по смыслу строк кода, находящихся в одном блоке.

Выравнивание делать табуляцией, а не пробелом.

Пример:

~~~bsl
ДатаСобытия			= ОбщегоНазначения._ТекущаяДатаСеанса();
Пакет				= ДанныеЛогаНазначения();
Очередь				= Пакет[Пакет.ВГраница() - 1].Выгрузить();
Слоты				= Пакет[Пакет.ВГраница()].Выбрать();
НазначаемаяЗадача	= ТаблицаЗадач[0];
~~~

## 2. Отступ между блоками кода, общих по смыслу

Пример:

~~~bsl
ДатаСобытия			= ОбщегоНазначения._ТекущаяДатаСеанса();
Пакет				= ДанныеЛогаНазначения();
Очередь				= Пакет[Пакет.ВГраница() - 1].Выгрузить();
Слоты				= Пакет[Пакет.ВГраница()].Выбрать();
НазначаемаяЗадача = ТаблицаЗадач[0];

Отбор = Новый Структура("Ссылка,ЗаказГруппы",
				НазначаемаяЗадача.Задача, НазначаемаяЗадача.Заказ);
СтрокиНазначения = Очередь.НайтиСтроки(Отбор);

Если СтрокиНазначения.Количество() = 0 Тогда
	ВызватьИсключение "Назначаемая задача отсутствует в очереди на момент сбора лога";
КонецЕсли;

ДанныеЗадачи = СтрокиНазначения[0];
Контекст = Новый Структура("Timestamp1c,TaskId,НазначаемаяЗадача",
			Формат(ДатаСобытия, "ДФ='yyyy-MM-dd HH:mm:ss.000'"),
			Строка(НазначаемаяЗадача.Задача.УникальныйИдентификатор()), НомерЗадачиДляЛога(ДанныеЗадачи.НомерЗадачи));
Контекст.Вставить("TaskNumber", ДанныеЗадачи.НомерДокумента);

СтрокиЛога = Новый Массив;
~~~

## 3. Отступы между логическими выражениями

Пример:

~~~bsl
Сообщение = НовоеСообщениеЛогаНазначения(Контекст, "ГрейлогНазначемаяЗадача");
Сообщение.Вставить("ГлавнаяЗадача"		, НомерЗадачиДляЛога(ДанныеЗадачи.ГлавнаяЗадача));
Сообщение.Вставить("Заказ"				, ДанныеЗадачи.Заказ);
Сообщение.Вставить("ЗаказГлавнойЗадачи"	, ДанныеЗадачи.ЗаказГлавнойЗадачи);

Если СтрокиНазначения.Количество() > 1 Тогда

	Сообщение.ГлавнаяЗадача = Null;
	Сообщение.ЗаказГлавнойЗадачи = Null;

КонецЕсли;
~~~

## 5. Формирование конструкторов выносить в отдельную функцию в область #Служебные процедуры и функции

Пример:

НЕПРАВИЛЬНО:

~~~bsl
КонецЦикла;

ПоляМетрик = Новый Структура("STARTW,Приоритет,AGEW,PAR_W,H0,H1,H2,H3,Total",
	"БонусЗаДобиваниеЗаказа", "УровеньПриоритета", "ШтрафЗаВозрастЗаказа",
	"БалансировочныйШтрафПерекоса", "H0", "H1", "H2", "H3", "ИтоговаяСтоимостьЗадачи");

Для Каждого ДанныеЗадачи Из Очередь Цикл

// ПРАВИЛЬНО:
ПоляМетрик = НовыйПоляМетрик()
//

Функция НовыйПоляМетрик

	ПоляМетрик = Новый Структура();
	ПоляМетрик.Вставить("STARTW"	, 0);
	ПоляМетрик.Вставить("Приоритет"	, 0)
	// т.д.

	Возврат ПоляМетрик;

КонецФункции
~~~

## 6. Наименование процедур и функций в соответсвии со стандартами 1С.

НЕПРАВИЛЬНО

~~~bsl
ЛогНазначения = ПодготовитьЛогНазначения(ТаблицаЗадач, ДанныеРабочегоМеста);

// ПРАВИЛЬНО
ЛогНазначения = ЛогНазначения(ТаблицаЗадач, ДанныеРабочегоМеста);
// НЕПРАВИЛЬНО
РассчитаемКоэффициентИзМетрики(Метрика)
// ПРАВИЛЬНО
РассчитатьКоэффициент(Метрика);
~~~

## 8. Если запрос выполняется к одному объекту метаданных то располагать этот запрос в модуле менеджера этого объекта

Пример:

~~~bsl
ВыборкаЗаказы = РегистрыСведений.ОформленныеЗаказа.Заказы(Данные);
// МодульМенеджера ргеистра
#Область ПрограммныйИнтерфейс
Функция ПолучитьЗаказы(Данные)
	Запрос = Новый Запрос;
	Запрос.Текст = "ВЫБРАТЬ Заказ ИЗ РегистрСведений.ОформленныеЗаказы";
	Результат = Запрос.Выполнить();
	Выборка = Результат.Выбрать();

	Возврат Выборка;

КонецФункции
#КонецОбласти
~~~

## 9. Если запрос состоит из различных объектов метаданных его так же необходимо выносить в отдельную процедуру и общий модуль

близкий по смыслу (или в текущий откуда вызывается)

ПРИМЕР

~~~bsl
Данные = ДанныеСотрудника(Сотрудник);

#Область ПрограмныйИнтерфейс
Функция ДанныеСотрудника(Сотрудник)

	Запрос = Новый Запрос;
	Запрос.Текст = "ВЫБРАТЬ Ссылка,
	| Должность
	| ИЗ
	| Справочник.Сотрудники КАК Сотрудники
	|	 ВНУТРЕННЕЕ СОЕДИНЕНИЕ Справочник.Должности КАК Должности
	|		ПО Сотрудники.Ссылка = Должности.Владелец";
	Результат = Запрос.Выполнить();
	Выборка = Результат.Выбрать();

	Возврат Выборка;

КонецФункции
#КонецОбласти
~~~
